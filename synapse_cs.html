/* C99 Hoare Partition QuickSort — In-Place Pointer Swap */
void quicksort_c99(int * restrict arr, int lo, int hi) {
  if (lo >= hi) return;
  int pivot = arr[lo + ((hi - lo) >> 1)], i = lo - 1, j = hi + 1;
  for (;;) {
    do { i++; } while (arr[i] < pivot);
    do { j--; } while (arr[j] > pivot);
    if (i >= j) break;
    int tmp = arr[i]; arr[i] = arr[j]; arr[j] = tmp;
  }
  quicksort_c99(arr, lo, j);
  quicksort_c99(arr, j + 1, hi);
}
